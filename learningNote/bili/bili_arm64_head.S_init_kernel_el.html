1. 概述：init_kernel_el 的使命

调用位置

// primary_entry (line 119-121)
1:  mov     x0, x19               // x0 = boot MMU state
    bl      init_kernel_el        // ★ 进入
    mov     x20, x0               // 保存 cpu_boot_mode

核心使命

  • 检测当前在 EL1 还是 EL2
  • 配置 SCTLR_ELx 为 MMU OFF 的安全状态
  • 通过 ERET 指令降级到 EL1
  • 返回 cpu_boot_mode（记录原始启动级别）
  • 如果从 EL2 进入：设置 HCR、VBAR、hypervisor stub
// ============ 两条路径统一目标 ============ 进入时 (CurrentEL) 离开时 (ERET 后) ───────────────── ──────────────── EL1 ──────────────────▶ EL1 (MMU off, 干净状态) EL2 ──────────────────▶ EL2 配置完成 → ERET → EL1 (MMU off) 无论从哪一级进入，init_kernel_el 返回后 CPU 一定在 EL1 这就是 "init_kernel_el" (Initialize Kernel Exception Level) 的含义
💡 为什么必须降级？ Linux 内核主体运行在 EL1。如果 bootloader 把 CPU 留在了 EL2（如 QEMU 默认行为），内核必须先接管 EL2 的控制权（设置 hypervisor stub），然后降级到 EL1 继续执行。如果已经在 EL1，只需配置好寄存器即可。

2. 完整源码

/*
 * Initialize the CPU exception level.
 * x0: MMU state at boot (0=cold, 1=warm)
 */
SYM_FUNC_START(init_kernel_el)
    mrs     x1, CurrentEL              // ① 读取当前异常级别
    cmp     x1, #CurrentEL_EL2         // ② 是否在 EL2?
    b.eq    init_el2                    // ③ 是 → 跳 EL2 路径

SYM_INNER_LABEL(init_el1, SYM_L_LOCAL)
    mov_q   x0, INIT_SCTLR_EL1_MMU_OFF // EL1 路径: 准备 SCTLR_EL1
    pre_disable_mmu_workaround
    msr     sctlr_el1, x0              // 写入 SCTLR_EL1 (MMU off!)
    isb
    mov_q   x0, INIT_PSTATE_EL1        // 准备目标处理器状态
    msr     spsr_el1, x0               // 写入 SPSR_EL1
    msr     elr_el1, lr                // ELR = 返回地址
    mov     w0, #BOOT_CPU_MODE_EL1     // w0 = 0xe11
    eret                               // ★ "异常返回"到 EL1

SYM_INNER_LABEL(init_el2, SYM_L_LOCAL)
    msr     elr_el2, lr                // EL2 路径: 保存返回地址

    // 如果 MMU 是开的 (warm boot), clean HYP 代码到 PoC
    cbz     x0, 0f
    adrp    x0, __hyp_idmap_text_start
    adr_l   x1, __hyp_text_end
    adr_l   x2, dcache_clean_poc
    blr     x2

    mov_q   x0, INIT_SCTLR_EL2_MMU_OFF
    pre_disable_mmu_workaround
    msr     sctlr_el2, x0              // 写入 SCTLR_EL2 (MMU off!)
    isb
0:
    init_el2_hcr    HCR_HOST_NVHE_FLAGS // 配置 HCR_EL2 (启用 EL1)
    init_el2_state                       // 配置 EL2 系统寄存器

    /* 安装 hypervisor stub 向量 */
    adr_l   x0, __hyp_stub_vectors
    msr     vbar_el2, x0
    isb

    mov_q   x1, INIT_SCTLR_EL1_MMU_OFF  // 准备 SCTLR_EL1

    // ★ VHE 检测: HCR_EL2.E2H 位是否设置
    mrs     x0, hcr_el2
    and     x0, x0, #HCR_E2H
    cbz     x0, 2f                      // E2H=0 → 非 VHE

    /* VHE 路径: 写 SCTLR_EL12 */
    msr_s   SYS_SCTLR_EL12, x1
    mov     x2, #BOOT_CPU_FLAG_E2H
    b       3f

2:  /* 非 VHE 路径: 写 SCTLR_EL1 */
    msr     sctlr_el1, x1
    mov     x2, xzr
3:
    mov     x0, #INIT_PSTATE_EL1        // 目标处理器状态
    msr     spsr_el2, x0

    mov     w0, #BOOT_CPU_MODE_EL2      // w0 = 0xe12
    orr     x0, x0, x2                  // 附加 E2H flag (如果有)
    eret                               // ★ "异常返回"到 EL1
SYM_FUNC_END(init_kernel_el)

3. primary_entry 中的调用点

// primary_entry (line 101-121)
    cbnz    x19, 0f                    // x19=MMU state
    ...  // Cold Boot: D-cache invalidate
    b       1f
0:  ...  // Warm Boot: D-cache clean
1:  mov     x0, x19                    // ★ x0 = x19 = boot MMU flag
    bl      init_kernel_el             // w0 = cpu_boot_mode
    mov     x20, x0                    // ★ 保存到 x20 (callee-saved)

    bl      __cpu_setup                // 配置 TCR/MAIR/等
    b       __primary_switch           // 开启 MMU!
寄存器方向值含义
x0入参x19 的值0=冷启动(D-cache OFF), 1=热启动(D-cache ON)
w0返回值0xe11 或 0xe12BOOT_CPU_MODE_EL1 或 BOOT_CPU_MODE_EL2 (可能带 E2H flag)
x20保存cpu_boot_modecallee-saved, 后续 __cpu_setup / __primary_switch 使用

调用时序：它在整个启动流程中的位置

// ============ primary_entry 调用链 ============ primary_entry: bl record_mmu_state // x19 = boot MMU flag bl preserve_boot_args // 保存 FDT ... 创建 idmap 页表 ... bl __pi_create_init_idmap // 构建 identity mapping ... cache 维护 (inval or clean) ... mov x0, x19 // 传入 boot MMU flag bl init_kernel_el // ★ 切换 EL + 关 MMU + 返回 boot mode mov x20, x0 // 保存 cpu_boot_mode bl __cpu_setup // 配置 TCR/MAIR/SCTLR b __primary_switch // 开启 MMU → 进入虚拟地址世界
💡 为什么把 x19 传进去？ 如果进入时在 EL2 且 MMU 是开的（warm boot），EL2 路径需要 clean HYP 代码段（`__hyp_idmap_text_start` 到 `__hyp_text_end`）到 PoC，确保后续 MMU 关闭后 HYP stub 代码在内存中可用。这个逻辑和 primary_entry 中 clean idmap 代码的思路完全对称。

4. EL1 路径详解

// ============ init_el1: 已经在 EL1, 只需配置后 ERET ============ CurrentEL = EL1 │ ▼ ① 写 SCTLR_EL1 = INIT_SCTLR_EL1_MMU_OFF ┌─────────────────────────────────────────────────────────┐ │ ENDIAN_SET_EL1 | LSMAOE | nTLSMD | EIS | TSCXT | EOS │ │ (EE + E0E) │ │ ★ 注意: M=0, C=0, I=0 — MMU 和 cache 全部关闭 │ └─────────────────────────────────────────────────────────┘ │ ▼ ② 写 SPSR_EL1 = INIT_PSTATE_EL1 ┌─────────────────────────────────────────────────────────┐ │ D | A | I | F | MODE=EL1h │ │ 屏蔽所有异常 (DAIF=1111) + SP 用 EL1h │ └─────────────────────────────────────────────────────────┘ │ ▼ ③ 写 ELR_EL1 = LR (返回地址) │ ▼ ④ w0 = BOOT_CPU_MODE_EL1 = 0xe11 │ ▼ ⑤ ERET → 返回到 EL1, 使用 SPSR_EL1 → PSTATE, PC = ELR_EL1 ★ 不是真正的"异常返回", 而是利用 ERET 机制切换处理器状态

init_el1 的三重作用

作用通过什么效果
关闭 MMU写入 SCTLR_EL1 (M=0)确保后续 idmap 代码在物理地址下执行
屏蔽中断SPSR_EL1 的 DAIF=1111在 MMU 开启的关键窗口期不允许中断打搅
返回调用者ELR_EL1 = LR + ERET利用 ERET 做 controlled return, 顺便设置 PSTATE
💡 为什么用 ERET 而不是普通 RET？ 普通 ret 只恢复 PC，不改 PSTATE。ERET 同时恢复 PC（从 ELR）和 PSTATE（从 SPSR），可以一次性关闭所有异常屏蔽位。这是 ARM64 保证在进入 C 代码之前处于已知安全处理器状态的惯用手段。

5. EL2 路径详解 ★

// ============ init_el2: 从 EL2 降级到 EL1 ============ CurrentEL = EL2 │ ├─ 保存 ELR_EL2 = LR │ ├─ if (x0 != 0) // warm boot: MMU was on │ clean HYP code [__hyp_idmap_text_start .. __hyp_text_end) │ ├─ 写 SCTLR_EL2 = INIT_SCTLR_EL2_MMU_OFF │ ┌──────────────────────────┐ │ │ SCTLR_EL2_RES1 | EE │ (M=0 — MMU 关闭) │ └──────────────────────────┘ │ ├─ init_el2_hcr(HCR_HOST_NVHE_FLAGS) │ 配置 HCR_EL2: E2H=0, TGE=0, RW=1 (EL1 用 AArch64) │ ├─ init_el2_state → 配置 CPTR, CNTHCTL 等 │ ├─ 安装 VBAR_EL2 = __hyp_stub_vectors │ ★ 后续若陷入 EL2, 由 hypervisor stub 处理 │ ├─ 准备 SCTLR_EL1 = INIT_SCTLR_EL1_MMU_OFF │ ├─ ★ VHE 分支 ★ │ │ │ ├── E2H=1 (VHE 模式) │ │ 写 SCTLR_EL12 ← 直接操作 EL1 的 SCTLR │ │ x2 = BOOT_CPU_FLAG_E2H (bit 32) │ │ │ └── E2H=0 (非 VHE) │ 写 SCTLR_EL1 │ x2 = 0 │ ├─ 写 SPSR_EL2 = INIT_PSTATE_EL1 │ ├─ w0 = BOOT_CPU_MODE_EL2 = 0xe12 ├─ w0 |= x2 (附加 E2H flag) │ └─ ERET → 降级到 EL1

VHE (Virtualization Host Extensions) 检测

什么是 VHE？ ARMv8.1 引入的特性。当 HCR_EL2.E2H=1 时，EL2 硬件直接支持运行宿主 OS，不需要软件在 EL2/EL1 间切换。大多数操作自动重定向到 EL2 对应寄存器。

VHE 时的 SCTLR 写入：

msr_s SYS_SCTLR_EL12, x1
// SCTLR_EL12 = "EL1 视角的 SCTLR"
// 硬件自动映射到 EL2 的对应控制

非 VHE 时的 SCTLR 写入：

msr sctlr_el1, x1
// 在 EL2 直接写 EL1 的 SCTLR
// 降级后 EL1 读到配置好的值

E2H flag 的意义：

w0 = BOOT_CPU_MODE_EL2 // 0x00000000_00000e12
w0 |= BOOT_CPU_FLAG_E2H // 0x00000001_00000e12
// bit 32 = 1 标记此内核在 VHE 模式下启动

init_el2 中的 Cache Clean（warm boot）

EL2 路径有一个和 primary_entry 对称的 warm boot 逻辑：

// init_el2 中 (line 290-294)
cbz     x0, 0f
adrp    x0, __hyp_idmap_text_start
adr_l   x1, __hyp_text_end
adr_l   x2, dcache_clean_poc
blr     x2
// primary_entry 中 (line 114-117)
0:  adrp    x0, __idmap_text_start
    adr_l   x1, __idmap_text_end
    adr_l   x2, dcache_clean_poc
    blr     x2

两者完全对称：primary_entry clean 的是 idmap 代码，init_el2 clean 的是 HYP 代码。都是因为后续要关闭 MMU，必须先把 cacheable 代码写回内存。

6. 返回值 x20 的去向

init_kernel_el 返回后，w0 = cpu_boot_mode 被保存到 x20（callee-saved 寄存器）。后续它在三处被使用：

使用点代码说明
__cpu_setup mov x0, x20 (隐式) 根据 boot mode 决定是否启用某些 CPU feature
__primary_switch 设置 TTBR1_EL1 等 启动 MMU，切换到虚拟地址空间
__primary_switched → C 代码 finalise_el2 如果 `is_hyp_mode_available()` 返回 true，安装 KVM
// ============ cpu_boot_mode 的最终用途 ============ init_kernel_el 返回 w0 = cpu_boot_mode │ ▼ x20 = cpu_boot_mode (0xe11 or 0xe12) │ ├── __cpu_setup 使用 x20 → 配置 TCR/MAIR │ └── __primary_switch 之后: __boot_cpu_mode[0] = BOOT_CPU_MODE_EL1 __boot_cpu_mode[1] = cpu_boot_mode ← 全局可见! 后续 C 代码 (setup_arch → finalise_el2): if (is_hyp_mode_available()) // __boot_cpu_mode[0] == BOOT_CPU_MODE_EL2 && // __boot_cpu_mode[1] == BOOT_CPU_MODE_EL2 // → 原始启动级别是 EL2 → 可以跑 KVM enable_kvm()
💡 为什么用 x20 而不是 x19？ x19 已经被用于存储 MMU boot state（0 或 1），而且在 cache 维护之后就已经消费完了。但 cpu_boot_mode 需要一直保留到 MMU 开启并进入 C 代码之后。x20 同样是 callee-saved 寄存器，和 x19、x21（FDT 指针）一样，可以安全穿越所有函数调用。

7.2 典型调试输出 — QEMU 默认启动 (EL2 → EL1)

QEMU 的 -M virt 默认让 CPU 从 EL2 启动，所以这条路径最常见：

(gdb) b init_kernel_el
(gdb) c
Breakpoint 1, init_kernel_el (x0=0x0) at head.S:270

(gdb) info registers x0 x1
x0   0x0    0        // Cold boot (D-cache was OFF)
x1   0x0    (will be overwritten)

(gdb) si                  // mrs x1, CurrentEL
(gdb) si                  // cmp x1, #CurrentEL_EL2
(gdb) p/x $x1
$1 = 0x8                 // CurrentEL = 0b10_00 = EL2 ✓

(gdb) si                  // b.eq init_el2 → taken!

// === 现在在 init_el2 路径 ===
(gdb) advance init_el2    // 跳到 init_el2 第一条

// 单步到 SCTLR_EL2 写入后
(gdb) until *0x41XXXXXX   // 跳过 warm boot check (x0=0, 跳过 clean)
(gdb) si                  // mov_q + msr sctlr_el2
(gdb) p/x $x0
$2 = 0x30c50838          // INIT_SCTLR_EL2_MMU_OFF

// 单步到 SCTLR_EL1 写入后 (非 VHE 路径)
(gdb) until *<eret_addr>
(gdb) p/x $x0
$3 = 0xe12               // BOOT_CPU_MODE_EL2

// ERET 执行后
(gdb) si                  // eret → 瞬间到 EL1
(gdb) info registers spsr_el1 currentel
SPSR_EL1    0x3c5        // D=1,A=1,I=1,F=1,M=EL1h
CurrentEL   0x4          // EL1 ✓

// 返回 primary_entry
(gdb) finish
(gdb) info registers x20
x20  0xe12                // cpu_boot_mode = BOOT_CPU_MODE_EL2 ✓

7.3 EL1 路径调试 (U-Boot 场景)

如果 bootloader 已经把 CPU 留在 EL1（某些 U-Boot 配置），会走 init_el1：

(gdb) b init_el1
(gdb) c
Breakpoint 2, init_el1 at head.S:275

(gdb) info registers currentel
CurrentEL   0x4          // EL1 ✓

(gdb) si                  // mov_q x0, INIT_SCTLR_EL1_MMU_OFF
(gdb) p/x $x0
$1 = 0x34d5d91d          // SCTLR_EL1 配置值 (M=0, C=0, I=0)

(gdb) si                  // msr sctlr_el1, x0

// 验证 SCTLR 确实被写入了
(gdb) p/x $SCTLR_EL1
$2 = 0x34d5d91d          // ★ 注意 GDB 只能在 MMU on 时读系统寄存器
                          // MMU off 时某些仿真器可能不支持

// 到 ERET 之前
(gdb) until *<eret_addr>
(gdb) p/x $SPSR_EL1
$3 = 0x3c5               // DAIF masked + EL1h mode
(gdb) p/x $ELR_EL1
$4 = 0x4163c114          // 返回地址 (primary_entry 中 bl 的下一行)

// w0 的值
(gdb) p/x $w0
$5 = 0xe11               // BOOT_CPU_MODE_EL1 ✓

define cpu_mode
    if $x20 == 0xe11
        printf "Boot mode: EL1\n"
    else
        if $x20 == 0xe12
            printf "Boot mode: EL2 (non-VHE)\n"
        else
            if ($x20 >> 32) & 1

end

7.5 验证清单

检查项方法预期
CurrentEL 检测在 b.eq init_el2 断下, 读 x1QEMU: 0x8 (EL2), U-Boot: 可能 0x4 (EL1)
SCTLR_ELx 写入 M=0在 msr sctlr_elx 后读寄存器bit 0 = 0 (MMU off)
Warm boot HYP clean设 x0=1 重跑, 确认 cbz 不跳 → 执行 clean只有 x0=1 时走 clean 分支
ERET 后 EL 切换si 过 eret, 读 CurrentEL0x4 (EL1), 不管原来在哪
SPSR 的 DAIF 全屏蔽读 SPSR_EL1 或 SPSR_EL2 的 bits[9:6]0b1111 (D=1, A=1, I=1, F=1)
w0 = 0xe11 或 0xe12在返回前读 w00xe11 (EL1) 或 0xe12 (EL2)
VHE 标志位读 w0 的 bit 32VHE 硬件: bit32=1; 非 VHE: bit32=0
⚠ 注意：MMU 关闭期间，GDB 通过 -s -S 访问物理内存中的数据是可行的，但系统寄存器（如 SCTLR_EL1）的读取取决于仿真器实现。QEMU 的 GDB stub 在 MMU 关闭时仍然可以读取系统寄存器，但某些硬件调试器可能不行。此时可以改在写 msr 指令之前读 x0 的值来间接验证。

8. 总结与关键要点

#要点说明
1统一出口：必定在 EL1无论从 EL1 还是 EL2 进入，ERET 后 CPU 一定在 EL1 执行。这是"init kernel EL"的核心语义
2三重任务关闭 MMU(写 SCTLR) + 屏蔽中断(通过 SPSR) + 降级 EL(通过 ERET)，一条 ERET 指令完成状态切换
3ERET 不是"异常返回"虽然叫"异常返回"，但这里并非从异常中恢复。init_kernel_el 主动构造 SPSR/ELR，利用 ERET 的 CPU 状态切换能力做受控降级。这是 ARM64 核心启动的惯用技巧
4EL2 对称的 Cache Clean如果 warm boot (MMU on)，EL2 路径会 clean HYP 代码段到 PoC，和 primary_entry 中 clean idmap 代码的逻辑完全对称
5VHE 感知通过 HCR_EL2.E2H 检测 VHE 模式，选择写 SCTLR_EL12 还是 SCTLR_EL1，并将 E2H flag 编码进返回值
6cpu_boot_mode 全局化返回值通过 x20 传递，最终存入全局数组 __boot_cpu_mode[]，后续 C 代码据此判断是否可在 EL2 运行 KVM
7SCTLR 写两次的用意EL2 路径中，先写 SCTLR_EL2（关 EL2 的 MMU），后写 SCTLR_EL1（关 EL1 的 MMU）。两次都清零 M 位，确保 EL1 和 EL2 的 stage 1 MMU 都关闭
