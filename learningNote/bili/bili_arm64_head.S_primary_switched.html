1. 概述：虚拟地址世界的入口

__primary_switched 是内核第一段跑在虚拟地址空间的代码。 上一段 __primary_switch 的最后两行 ldr x8, =__primary_switched; br x8 把 PC 从 物理地址（idmap）跳到了高 VA。到这里为止, MMU 已开、TTBR1 已指向 swapper_pg_dir, 内核真正"活"在了 0xffff8000... 的虚拟地址里。

// ============ 调用链与寄存器入参 ============ primary_entry → __cpu_setup → __primary_switch → __enable_mmu (MMU ON) → __pi_early_map_kernel (建高 VA 页表 + 重定位 + 装 TTBR1) ldr x8, =__primary_switched br x8 ★ 跳到虚拟地址世界! __primary_switched 入参 (从 primary_entry 一路保存下来): x0 = __pa(KERNEL_START) = _text 的物理地址 x20 = cpu_boot_mode (BOOT_CPU_MODE_EL1 / EL2) x21 = FDT 物理地址指针 注释 (head.S:218): "executed with the MMU enabled" —— 此函数全程 MMU 开启

它的使命是"架桥": 把汇编阶段攒下的关键信息(启动模式、FDT)落到全局变量, 初始化 C 代码运行所需的一切(current、异常向量表、per-cpu 偏移), 最后调用 C 世界的 start_kernel()。

2. __primary_switched 完整源码 (head.S:220-248)

SYM_FUNC_START_LOCAL(__primary_switched)
    adr_l   x4, init_task                // ① 取 init_task 地址
    init_cpu_task x4, x5, x6             // ② 初始化 CPU 任务上下文 (current/栈/FP)

    adr_l   x8, vectors                  // ③ 加载异常向量表 (虚拟地址)
    msr     vbar_el1, x8                 //    安装到 VBAR_EL1
    isb

    stp     x29, x30, [sp, #-16]!        // ④ 标准 C 函数序言 (压 FP/LR)
    mov     x29, sp

    str_l   x21, __fdt_pointer, x5       // ⑤ 全局保存 FDT 指针

    adrp    x4, _text                    // ⑥ ★ 计算 kimage_voffset
    sub     x4, x4, x0                   //    = VA(_text) - PA(_text)
    str_l   x4, kimage_voffset, x5       //    保存到全局变量

    mov     x0, x20
    bl      set_cpu_boot_mode_flag       // ⑦ 记录 boot CPU mode

#if defined(CONFIG_KASAN_GENERIC) || defined(CONFIG_KASAN_SW_TAGS)
    bl      kasan_early_init             // ⑧ KASAN 早期初始化 (可选)
#endif
    mov     x0, x20
    bl      finalise_el2                 // ⑨ 尝试启用 VHE (EL2 模式)

    ldp     x29, x30, [sp], #16          // ⑩ 恢复栈帧
    bl      start_kernel                 // ⑪ ★ 进入 C 代码主函数!
    ASM_BUG()                            // ⑫ start_kernel 不返回, 永远不会到这
SYM_FUNC_END(__primary_switched)

3. init_cpu_task：建立 init_task 上下文 (head.S:195-213)

这是 C 代码能否运行的根基:C 世界里 current 宏、内核栈、backtrace 全都依赖这几步。宏定义如下:

.macro  init_cpu_task tsk, tmp1, tmp2
    msr     sp_el0, \tsk                    // ① sp_el0 = init_task → current 可用

    ldr     \tmp1, [\tsk, #TSK_STACK]       // ② 读 init_task 的内核栈基址
    add     sp, \tmp1, #THREAD_SIZE         // ③ sp = 栈顶
    sub     sp, sp, #PT_REGS_SIZE           //    预留 pt_regs 空间

    stp     xzr, xzr, [sp, #S_STACKFRAME]   // ④ 清零 stackframe (fp, lr)
    mov     \tmp1, #FRAME_META_TYPE_FINAL   // ⑤ 标记为 FINAL 帧
    str     \tmp1, [sp, #S_STACKFRAME_TYPE] //    (unwinder 的终止条件)
    add     x29, sp, #S_STACKFRAME          // ⑥ FP = 帧记录地址

    scs_load_current                        // ⑦ 加载影子调用栈指针 (若启用)

    adr_l   \tmp1, __per_cpu_offset         // ⑧ 读本 CPU 的 per-cpu 偏移
    ldr     w\tmp2, [\tsk, #TSK_TI_CPU]     //    从 thread_info 拿 CPU 号
    ldr     \tmp1, [\tmp1, \tmp2, lsl #3]   //    offset = __per_cpu_offset[cpu]
    set_this_cpu_offset \tmp1               //    写入 tpidr_el1
.endm

逐行要点

行含义
①ARM64 的 current 存在 sp_el0 里。msr sp_el0, init_task 之后, C 代码里 current 宏读 sp_el0 就能拿到 init_task, 不必走 thread_info
②③从 init_task 的 stack 字段拿到栈底, 加 THREAD_SIZE 到栈顶, 再减去 PT_REGS_SIZE 预留一段 pt_regs 空间, 与用户任务/kthread 的布局保持一致
④⑤⑥造一个 FINAL 帧记录: 在栈顶放 fp=0, lr=0, 并把类型标记为 FRAME_META_TYPE_FINAL。这样 unwinder 走到这里就知道是栈底, backtrace 不会越过它继续往非法地址解引用
⑦scs_load_current: 若启用了影子调用栈 (SCS), 把 SCS 指针装进 x18
⑧per-cpu 偏移: 从 thread_info.cpu 拿到 CPU 号, 索引 __per_cpu_offset[], 把该 CPU 的偏移写到 tpidr_el1。per_cpu_ptr() 宏就靠这个偏移换算每个 CPU 的变量地址
💡 为什么此时才设置这些？ 因为 init_cpu_task 访问的都是虚拟地址 (init_task、__per_cpu_offset、栈), 只有在 MMU 开启、内核映射到高 VA 之后才能安全访问。这正是它出现在 __primary_switched 最开头的原因。

3.1 预留 pt_regs 空间：sub sp, sp, #PT_REGS_SIZE

sub sp, sp, #PT_REGS_SIZE 就是把栈指针往下挪 sizeof(struct pt_regs) 字节, 在内核栈最顶端预留一整块 struct pt_regs。

先看栈是怎么一步步设出来的:

ldr     \tmp1, [\tsk, #TSK_STACK]     // tmp1 = init_task 的栈底地址
add     sp, \tmp1, #THREAD_SIZE       // sp = 栈顶 (栈底 + THREAD_SIZE)
sub     sp, sp, #PT_REGS_SIZE         // sp 下移, 预留 pt_regs 空间 ★

栈是向下生长的, 而 ARM64 的内核栈约定是:struct pt_regs 固定放在栈的最顶端。所以先让 sp 指到栈顶, 再往下挪一个 pt_regs 的大小, 空出的这块就是 pt_regs 的位置。

pt_regs 是什么

struct pt_regs ([ptrace.h:156](arch/arm64/include/asm/ptrace.h#L156)) 是 CPU 在异常/系统调用时保存全套寄存器的固定布局:

struct pt_regs {
    union {
        struct user_pt_regs user_regs;
        struct {
            u64 regs[31];   // x0..x30
            u64 sp;
            u64 pc;
            u64 pstate;
        };
    };
    u64 orig_x0;
    s32 syscallno; u32 pmr;
    u64 sdei_ttbr1;
    struct frame_record_meta stackframe;   // ← fp + pc/type
};

栈布局

// ============ init_task 内核栈布局 ============ 高地址 +------------------------+ stack_base + THREAD_SIZE ← 栈顶 | | | struct pt_regs | ← sub 之后 sp 指到这块底部 | (x0..x30, sp, pc, | [栈顶-PT_REGS_SIZE, 栈顶) | pstate, stackframe) | +------------------------+ ← sp 现在指向这里 | | | 实际可用栈空间 | ← 后续 push 从这里继续向下 | (向下生长) | | | +------------------------+ stack_base ← 栈底 (TSK_STACK) 低地址 这对应宏 task_pt_regs(task) = (struct pt_regs *)(THREAD_SIZE + task_stack_page(task)) - 1

为什么每个任务都要预留

  1. 给 unwinder 一个可识别的"最终帧" — backtrace 靠 task_pt_regs(current) 这个固定位置找最底层帧记录。这里把 pt_regs.stackframe 清零并标 FRAME_META_TYPE_FINAL, unwinder 走到这知道是栈底, 不会越界往非法地址解引用。
  2. 与用户任务/kthread 保持一致 — 用户进程进内核时它的 pt_regs 就放在栈顶 (存被保存的用户态寄存器)。init_task 永远不会有"从用户态进来", 但同样预留, 让 task_pt_regs(current) 这个通用宏对任何任务都成立, 不用特判。

3.2 scs_load_current：影子调用栈 (SCS)

scs_load_current 是 SCS (Shadow Call Stack, 影子调用栈) 的初始化: 把当前任务的 SCS 指针装进 x18。展开后 (arch/arm64/include/asm/scs.h:13-16):

#ifdef CONFIG_SHADOW_CALL_STACK
    scs_sp  .req    x18                 // 声明 x18 为 SCS 指针

    .macro scs_load_current
    get_current_task scs_sp              // ① x18 = current (读 sp_el0)
    ldr     scs_sp, [scs_sp, #TSK_TI_SCS_SP]  // ② x18 = current->thread_info.scs_sp
    .endm
#else
    .macro scs_load_current              // 未启用 SCS → 空宏, 无操作
    .endm
#endif

SCS 是什么

影子调用栈是一种防 ROP (Return-Oriented Programming) 的安全机制(Clang 提出, Android/arm64 上大量使用)。核心思想:

// ============ 普通栈 vs 影子调用栈 ============ 普通调用: bl foo → 返回地址 x30 压到普通栈 普通栈可能被缓冲区溢出覆盖 → 攻击者改写返回地址 → ROP SCS 方案: 函数序言: str x30, [x18]; add x18, x18, #8 // 返回地址额外压到影子栈 函数尾声: ldr x30, [x18, #-8]; sub x18, x18, #8 x18 指向的是一块独立内存 (影子栈), 只存返回地址 普通栈溢出碰不到影子栈 → 返回地址安全

关键在于 x18 是 ARM64 的平台寄存器 (platform register), 被约定专门留给 SCS 指针使用。编译器 (Clang -fsanitize=shadow-call-stack) 会在每个函数的序言/尾声自动插入上面的 push/pop 指令, 无需手写。

两步拆解

步指令含义
①get_current_task scs_sp = mrs x18, sp_el0拿到 current 任务指针 (前面已 msr sp_el0, init_task)
②ldr x18, [x18, #TSK_TI_SCS_SP]读 thread_info.scs_sp 字段, 得到该任务自己的影子栈指针, 装进 x18

每个任务都有自己独立的影子栈, 指针存在 thread_info.scs_sp 里 (上下文切换时通过 scs_save/scs_load 保存恢复)。init_task 的初始值指向静态分配的 init_shadow_call_stack ([thread_info.h:122](arch/arm64/include/asm/thread_info.h#L122))。

💡 两种 SCS 别混淆: 这里 scs_load_current 是编译期 SCS (CONFIG_SHADOW_CALL_STACK, 编译器直接生成 push/pop)。 与之相对的是 map_kernel.c 里的 scs_patch() —— 那是动态 SCS (CONFIG_UNWIND_PATCH_PAC_INTO_SCS), 在没有 PAC/BTI 的机器上启动时把 paciasp/autiasp 指令补丁成 SCS push/pop。两者都是为了让返回地址不进普通栈。

4. 安装异常向量表 VBAR_EL1

    adr_l   x8, vectors        // 加载 vectors 的虚拟地址
    msr     vbar_el1, x8       // 写 VBAR_EL1
    isb

这是极其关键的一步:在此之前(整个汇编启动阶段), VBAR_EL1 要么是 0、要么是无效值, 系统没有可用的异常向量表。

而 __enable_mmu 之后 MMU 已经开启, 任何内存访问都要经过页表翻译——一旦出现缺页、取指异常、中断, CPU 会跳到 VBAR 指向的向量。如果 VBAR 没装好, 异常就是未定义行为。

⚠️ MMU ON → VBAR 装好 之间的空窗期: 从 set_sctlr_el1 (MMU 开启) 到这条 msr vbar_el1, 中间如果发生异常, 后果不可预知。 所以 __primary_switched 一进来就赶紧装向量表, 把这个空窗缩到最小。

装好之后, 内核就能处理 IRQ、缺页、系统调用等异常了。这也是为什么 start_kernel() 之前必须先到这里。

5. 保存 FDT 指针 __fdt_pointer

    str_l   x21, __fdt_pointer, x5   // 把 FDT 物理地址写入全局变量

x21 是启动时从 preserve_boot_args 一路保存下来的 FDT (设备树) 物理地址。 这里把它落到全局变量 __fdt_pointer, 之后 setup_arch() 靠它解析设备树(内存布局、CPU 拓扑、外设信息等)。

注意存的是物理地址: 设备树此时还在低物理内存里, 由 idmap 映射, 尚未建立独立映射。内核后续会自行处理它的访问。

6. kimage_voffset：VA-PA 偏移 ★

    adrp    x4, _text            // x4 = VA(_text)
    sub     x4, x4, x0           // x0 = PA(_text) → x4 = VA - PA
    str_l   x4, kimage_voffset, x5

这是整个内核最重要的运行时常量之一。它记录"内核虚拟地址与物理地址之间的固定偏移", 之后任意内核符号都可以在 VA 和 PA 之间互转:

// ============ kimage_voffset 的两个用法 ============ include/asm/memory.h: #define __kimg_to_phys(addr) ((addr) - kimage_voffset) // VA → PA #define __phys_to_kimg(x) ((unsigned long)((x) + kimage_voffset)) // PA → VA __pa(x) / __va(x) 等宏最终都落到这两个公式上

数值示例 (QEMU, 无 KASLR)

// ============ 计算过程 ============ adrp x4, _text // 此刻在 VA 空间, adrp 得到 _text 的链接地址 // x4 = 0xffff800080000000 x0 = __pa(_text) // __primary_switch 传进来的物理地址 // x0 = 0x40200000 sub x4, x4, x0 // kimage_voffset // = 0xffff800080000000 - 0x40200000 // = 0xffff80003FE00000
💡 有 KASLR 时: adrp x4, _text 得到的是 KIMAGE_VADDR + kaslr_offset, 所以 kimage_voffset = (KIMAGE_VADDR + kaslr_offset) - pa_base。KASLR 的位移被"吸收"进了这个偏移里, 这也是为什么 kimage_voffset 是随机值、每台机器/每次启动都不同。

为什么此时算, 不在更早算？

要算 VA(_text), 必须让 adrp 在虚拟地址空间里解析 _text。 只有 MMU 开启、内核映射到高 VA 之后, adrp x4, _text 才会得到链接时的 VA。 在之前的 idmap(物理地址)阶段, 同样一条 adrp 得到的是物理地址, 减出来是 0, 没意义。

7. boot mode / VHE / start_kernel

⑦ set_cpu_boot_mode_flag (head.S:418-425)

SYM_FUNC_START_LOCAL(set_cpu_boot_mode_flag)
    adr_l   x1, __boot_cpu_mode
    cmp     w0, #BOOT_CPU_MODE_EL2
    b.ne    1f
    add     x1, x1, #4
1:  str     w0, [x1]            // __boot_cpu_mode[0]=EL1 或 __boot_cpu_mode[1]=EL2
    ret
SYM_FUNC_END(set_cpu_boot_mode_flag)

把启动时的 CPU 模式 (EL1 还是 EL2) 记进 __boot_cpu_mode[]。之后内核需要知道"是否从 EL2 启动"来决策 KVM、VHE 等, 就查这个数组。

⑨ finalise_el2：VHE (hyp-stub.S:244-258)

如果内核是从 EL2 启动的, 这里做两件事:

  • 硬件支持 VHE → 启用 VHE, 让内核以 EL2 的宿主权限运行(异常向量、系统寄存器都按 EL2 解释);
  • 不支持 VHE → 把 CPU 降到 EL1, 并初始化 EL2 的 stub (供之后 KVM 用)。

⑩⑪ bl start_kernel

    ldp     x29, x30, [sp], #16   // 恢复栈帧
    bl      start_kernel          // 进入 C 世界
    ASM_BUG()                     // start_kernel 永不返回

bl start_kernel 之后, 控制权交给 init/main.c 的 C 代码, 汇编启动流程正式结束。 start_kernel() 是一个 __noreturn 函数, 如果它真返回了, ASM_BUG() 会触发一个明确崩溃, 而不是静默跑到未知地址。

8.2 物理/虚拟地址分界

⚠️ 调试时最容易踩的坑: __primary_switched 是物理地址与虚拟地址的分界点。 它之前的代码 (含 __enable_mmu) 跑在物理地址, 它及之后 (含 start_kernel) 跑在虚拟地址。 如果你的 GDB 脚本 (如 fix_vmlinux_head_entry.sh) 把所有符号按固定公式重定位到物理地址来设断点, 那么从 __primary_switched 开始这些断点就再也打不中了——因为运行时 PC 已经是高 VA。 此时要改用虚拟地址断点, 例如: 
hbreak *0xffff800081450380          # start_kernel 的虚拟地址 (硬件断点)
symbol-file vmlinux
break start_kernel                  # 或者用虚拟符号直接断
