1. 方案概述

// ============ AI GDB 调试工作流 ============ ┌──────────┐ GDB脚本 ┌──────────┐ 实时输出 ┌──────────┐ │ AI │ ──────────────▶ │ 用户 │ ──────────────▶ │ 用户 │ │ 编写脚本 │ │ 运行 │ │ 观察 │ └──────────┘ └──────────┘ └──────────┘ ▲ │ │ 复制输出 │ └───────────────────────────────────────────────────────────┘ 一轮流程: AI写脚本 → 用户运行 → 用户看输出 → 贴给AI → AI分析 → 下一轮

核心思路

  • AI 负责写 GDB 命令 — 不需要交互式输入，不需要 `expect`，只需把命令写入文件
  • 用户负责运行 — 一条命令 `gdb-multiarch -nx -batch -x script.gdb`，所有步骤在一个会话内串行完成
  • `shell sleep N` 创建暂停 — 用户有时间看清每一步的输出
  • 每次都重启 QEMU — batch 模式退出后 QEMU 继续跑，下次需要从头开始

2. 环境准备

2.1 启动 QEMU（冻结 CPU，不连 GDB）

cd ~/kernel/linux-6.18.1

qemu-system-aarch64 -s -S \
    -machine virt -cpu cortex-a57 -m 1024 -smp 4 \
    -kernel arch/arm64/boot/Image \
    --append "nokaslr rdinit=/linuxrc console=ttyAMA0" \
    -nographic &

# -s : 监听 GDB stub 端口 :1234
# -S : 冻结 CPU 在第一条指令，等待 GDB 连接
# nokaslr : 禁用 KASLR，保证断点地址每次一致

2.2 关键地址（固定不变，nokaslr 保证）

地址符号说明
0x40000000QEMU bootloader6 条指令加载内核入口 → br x4
0x40200000_textefi_signature_nop + b primary_entry
0x4163c0f0primary_entry启动主入口（init.text 段）
0x4163c0f4primary_entry+4★ bl record_mmu_state 返回后
0x4163c110primary_entry+32★ bl create_init_idmap 返回后
0x4163c150primary_entry+96★ bl init_kernel_el 返回后 (mov x20,x0)

2.3 获得精确地址的方法

# 方法1: 先跑一次 disasm 探测
gdb-multiarch -nx -batch \
  -ex "target remote :1234" \
  -ex "add-symbol-file ..." \
  -ex "b *0x40200000" -ex "c" \
  -ex "advance primary_entry" \
  -ex "x/30i 0x4163c0f0" \
  -ex "detach" -ex "quit"

# 方法2: nm + 偏移计算
nm vmlinux | grep primary_entry
# → ffff80008143c0f0 T primary_entry
# PA = VA - 0xffff800080000000 + 0x40200000
# PA(primary_entry) = 0xffff80008143c0f0 - 0xffff800080000000 + 0x40200000
#                   = 0x4163c0f0

4. GDB 脚本模板

4.1 终极调试脚本（6 步全流程）

# 文件: /tmp/ai_final.gdb
# 运行: gdb-multiarch -nx -batch -x /tmp/ai_final.gdb

set pagination off
set confirm off
set architecture aarch64
target remote :1234
add-symbol-file /home/ybzhang/kernel/linux-6.18.1/vmlinux \
  -s .head.text 0x40200000 -s .text 0x40210000 -s .rodata 0x410f0000 \
  -s .init.text 0x41650000 -s .rodata.text 0x41637800 -s .init.data 0x41738000

echo ╔══════════════════════════════════════════════╗\n
echo ║  AI 自动调试 ARM64 head.S                    ║\n
echo ╚══════════════════════════════════════════════╝\n

echo \n━━━ 1/6 QEMU Bootloader @ 0x40000000 ━━━\n
x/6i 0x40000000
shell sleep 2

echo \n━━━ 2/6 _text 内核入口 @ 0x40200000 ━━━\n
b *0x40200000
c
x/2i $pc
shell sleep 2

echo \n━━━ 3/6 primary_entry ━━━\n
advance primary_entry
printf "FDT(x0)=0x%lx\n", $x0
shell sleep 2

echo \n━━━ 4/6 → bl record_mmu_state 后 ━━━\n
b *0x4163c0f4
c
printf "x19=%ld → %s\n", $x19, $x19?"Warm":"Cold"
shell sleep 2

echo \n━━━ 5/6 → bl create_init_idmap 后 ━━━\n
b *0x4163c110
c
printf "ptep=0x%lx  pages=%ld\n", $x0, ($x0-0x41730000)/4096
shell sleep 2

echo \n━━━ 6/6 → bl init_kernel_el 后 ━━━\n
b *0x4163c150
c
printf "cpu_boot_mode=0x%x (%s)\n", $w0, $w0==0xe12?"EL2":"EL1"
shell sleep 2

detach
quit

4.2 关键技术要点

技术命令作用
跳过 .gdbinit-nx避免 TUI 在 batch 模式报错，避免 auto-load 安全警告
手动加载符号add-symbol-file vmlinux -s .text 0x40210000 ...绕过 fix_vmlinux 脚本，直接用绝对路径和已知 PA 映射
精确地址断点b *0x4163c0f4物理地址断点，不依赖 GDB 符号解析
步骤间暂停shell sleep 2让用户看清每步输出，模拟交互式断点效果
advance 跳函数advance primary_entry跳过 bootloader，直接到 primary_entry
nokaslr--append "nokaslr ..."禁用内核地址随机化，保证每次断点地址相同

6. 踩坑记录与技巧

#问题现象根因修复
1 -batch 模式下 TUI 报错 Cannot enable the TUI when output is not a terminal ~/.gdbinit 里配置了 layout src 和 tui enable，batch 模式不支持 加 -nx 跳过 .gdbinit
2 QEMU 在 batch 之间继续跑 第二次 batch 连接时 PC 已经在很远的地址，断点失效 detach 后 QEMU 不暂停，CPU 继续执行 每次 batch 前重启 QEMU（fuser -k 1234; qemu ... &）
3 finish 报错 "finish" not meaningful in the outermost frame 断点命中在函数第一条指令，还没建立栈帧 用 b *addr 设在函数返回后的地址，不用 finish
4 符号查找失败 vmlinux: No such file or directory 当前工作目录不在内核根目录，相对路径找不到 vmlinux 使用绝对路径 /home/ybzhang/kernel/linux-6.18.1/vmlinux
5 $currentel>>2 算术报错 Argument to arithmetic operation not a number batch 模式下 $currentel 返回的不是数值类型 用 $x19>>2 代替（record_mmu_state 已把 CurrentEL 读到 x19）
6 端口占用 Failed to find an available port: Address already in use 上次 QEMU 未正常退出，端口残留 fuser -k 1234/tcp 强制释放端口

6.1 快速重启脚本

#!/bin/bash
# 一键重启 QEMU + 运行 AI 脚本
fuser -k 1234/tcp 2>/dev/null
sleep 1
cd ~/kernel/linux-6.18.1
qemu-system-aarch64 -s -S -machine virt -cpu cortex-a57 -m 1024 -smp 4 \
    -kernel arch/arm64/boot/Image \
    --append "nokaslr rdinit=/linuxrc console=ttyAMA0" -nographic &
sleep 1
gdb-multiarch -nx -batch -x /tmp/ai_final.gdb
