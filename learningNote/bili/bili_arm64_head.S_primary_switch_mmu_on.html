1. 概述：primary_entry 的最后一条指令

// ============ primary_entry 完整调用链 ============ primary_entry: bl record_mmu_state // x19 = boot MMU flag bl preserve_boot_args // 保存 FDT + boot_args ... 创建 idmap ... bl __pi_create_init_idmap // 构建 identity mapping 页表 ... cache 维护 ... mov x0, x19 bl init_kernel_el // EL 切换, 关 MMU → w0=cpu_boot_mode mov x20, x0 // x20 = cpu_boot_mode bl __cpu_setup // 配置 TCR/MAIR → x0=SCTLR_MMU_ON b __primary_switch // ★ 最后一条指令: 无条件跳转 // 此后 primary_entry 不再返回 __primary_switch: bl __enable_mmu // ★① 开启 MMU (idmap 生效) mov sp, early_init_stack // ② 切到新栈 (现在是 VA 了) bl __pi_early_map_kernel // ③ 重定位内核到高 VA ldr x8, =__primary_switched // ④ 加载绝对 VA (高地址) br x8 // ⑤ 跳转到虚拟地址世界!

从 b __primary_switch 到 bl start_kernel，总共约 20 条指令，完成了 ARM64 内核启动中最重要的三件事：开启 MMU、重定位内核、进入 C 代码。

2. __primary_switch 逐行分析 (head.S:508-523)

SYM_FUNC_START_LOCAL(__primary_switch)
    adrp    x1, reserved_pg_dir            // ① x1 = 空页表 (全零 PG)
    adrp    x2, __pi_init_idmap_pg_dir     // ② x2 = idmap PGD
    bl      __enable_mmu                   // ③ ★ 开启 MMU!

    // ===== 以下是 MMU ON 之后 =====
    adrp    x1, early_init_stack           // ④ 获取新栈地址 (现在是 VA!)
    mov     sp, x1                         // ⑤ 切到新栈
    mov     x29, xzr                       // ⑥ FP=0 (新栈帧起点)
    mov     x0, x20                        // ⑦ x0 = cpu_boot_mode
    mov     x1, x21                        // ⑧ x1 = FDT 物理地址
    bl      __pi_early_map_kernel          // ⑨ 重定位 + 映射内核

    ldr     x8, =__primary_switched        // ⑩ x8 = __primary_switched 的绝对 VA
    adrp    x0, KERNEL_START               // ⑪ x0 = _text 的物理地址
    br      x8                             // ⑫ 跳转到虚拟地址!
行指令含义
①②adrp x1, reserved_pg_dir
adrp x2, __pi_init_idmap_pg_dir		x1 = reserved 空页表 (全 0, 用于 TTBR1 切换)
x2 = 我们刚刚构建的 idmap PGD
③bl __enable_mmu★ 开启 MMU! 见下一章详解
④⑤adrp x1, early_init_stack; mov sp, x1重新设置栈指针。MMU ON 之前这行用的是物理地址，MMU ON 之后同一地址经过 idmap 翻译 → 结果相同
⑥mov x29, xzrFP=0, 新栈帧起点。GDB backtrace 的终止条件
⑦⑧mov x0, x20; mov x1, x21传递 cpu_boot_mode 和 FDT 给 __pi_early_map_kernel
⑨bl __pi_early_map_kernel映射内核到高 VA, 重定位页表, 设置 TTBR1
此后内核主体可以跑在高 VA 上了
⑩⑪⑫ldr x8, =__primary_switched
br x8		★ 这是从 idmap (物理地址) 到高 VA (虚拟地址) 的跳跃!
ldr x8, = 加载的是链接器分配的绝对 VA

关键：br x8 为什么能成功？

// ============ br x8 的时刻 ============ 执行 br x8 之前: PC = 某个物理地址 (通过 idmap VA=PA 翻译) 页表: TTBR0 = idmap PGD, TTBR1 = 内核正式 PGD 执行 br x8: x8 = 0xffff8000_xxxx_xxxx (__primary_switched 的链接 VA) CPU 发出地址 0xffff8000_xxxx_xxxx: → 这是高 VA 地址 → TCR_EL1.T1SZ 判定 → 走 TTBR1 翻译 → TTBR1 指向 __pi_early_map_kernel 刚建好的内核页表 → 翻译成功! CPU 跳到 __primary_switched 这一瞬间, CPU 从 "idmap 物理地址世界" 跳到了 "内核虚拟地址世界" 这是 ARM64 启动中最精彩的单条指令

3. __enable_mmu: MMU 开启的瞬间 (head.S:459-473)

SYM_FUNC_START(__enable_mmu)
    mrs     x3, ID_AA64MMFR0_EL1           // ① 读硬件特性寄存器
    ubfx    x3, x3, #ID_AA64MMFR0_EL1_TGRAN_SHIFT, 4
    cmp     x3, #ID_AA64MMFR0_EL1_TGRAN_SUPPORTED_MIN
    b.lt    __no_granule_support            // ② 4KB 粒度不支持? → 死循环
    cmp     x3, #ID_AA64MMFR0_EL1_TGRAN_SUPPORTED_MAX
    b.gt    __no_granule_support            // ③ 粒度编号超出范围? → 死循环
    phys_to_ttbr x2, x2                     // ④ 物理地址 → TTBR 格式
    msr     ttbr0_el1, x2                   // ⑤ 安装 idmap PGD 到 TTBR0
    load_ttbr1 x1, x1, x3                   // ⑥ 安装 reserved 空页表到 TTBR1

    set_sctlr_el1   x0                      // ⑦ ★ 写入 SCTLR_EL1 = MMU_ON!
                                            //    msr sctlr_el1, x0
                                            //    isb
                                            //    ic iallu     ← 刷 I-cache
                                            //    dsb nsh
                                            //    isb
    ret                                     // ⑧ ★ 返回, 但此时 MMU 已在翻译!
SYM_FUNC_END(__enable_mmu)

逐行解释

行含义
①②③安全检查：读 ID_AA64MMFR0_EL1.TGran4 确认硬件支持 4KB 页表粒度。不支持则跳入 __no_granule_support — 无限循环 (wfi/ b)
④phys_to_ttbr x2, x2：把 idmap PGD 的物理地址转换为 TTBR 格式。
对于 48-bit PA: 直接返回 PA (bits[47:1] ← PA[47:1], bit[0] 可能为 CnP)
对于 52-bit PA: 做位折叠 (类似 __phys_to_pte_val)
⑤msr ttbr0_el1, x2：安装 idmap PGD 到 TTBR0。
★ 此后硬件遍历页表时就会读我们的 idmap 页表了
⑥load_ttbr1 x1, x1, x3：安装 reserved 空页表到 TTBR1。
reserved_pg_dir 是一页全 0 的物理页 — 任何 TTBR1 翻译都产生 translation fault
⑦set_sctlr_el1 x0：这条宏做了三件事：
1. msr sctlr_el1, x0 — 写入 SCTLR (M=1!) ★ MMU 在此刻生效
2. isb — 同步
3. ic iallu + dsb nsh + isb — 全刷 I-cache (清除 speculative fetch 的旧指令)
⑧ret：返回 __primary_switch。
★ 这条 ret 本身也是经 MMU 翻译的!
取指: PC → idmap 查表 → PA=PC → 正确!
这就是 idmap 存在的全部意义

set_sctlr_el1 的展开

// arch/arm64/include/asm/assembler.h:720-731
.macro set_sctlr, sreg, reg
    msr     \sreg, \reg          // 写 SCTLR_EL1 = INIT_SCTLR_EL1_MMU_ON
    isb                          // ★ 指令同步屏障
    ic      iallu                // 全刷 I-cache 到 PoU
    dsb     nsh
    isb
.endm

ic iallu 刷新整个 I-cache 到 PoU (Point of Unification)。因为 MMU 开启改变了翻译环境，之前 speculative fetch 进来的旧指令可能基于错误的翻译。不刷 I-cache 可能导致 CPU 执行到旧翻译下的错误指令。

MMU 开启的精确时刻

// ============ msr sctlr_el1, x0 前后的变化 ============ msr sctlr_el1, x0 之前: CPU 取指: PA = PC (MMU off, 无翻译) 地址空间: 物理地址 msr sctlr_el1, x0 之后 (M 位从 0 → 1): CPU 取指: PA = idmap_walk(PC) = PC (VA==PA, identity 翻译) 地址空间: 仍然是物理地址 (因为 idmap 保证 VA==PA) TTBR0_EL1 指向的 idmap PGD 开始被硬件使用 TCR_EL1 的 T0SZ/TG0/SH0 等配置开始生效

4. MMU ON 之后：__pi_early_map_kernel 详解 (map_kernel.c)

    mov     x0, x20                        // cpu_boot_mode
    mov     x1, x21                        // FDT 物理地址
    bl      __pi_early_map_kernel          // 映射 + 重定位内核!

这里的 bl __pi_early_map_kernel 调用的不是汇编, 而是一个 C 函数 early_map_kernel() (定义于 arch/arm64/kernel/pi/map_kernel.c:241)。 它执行时 MMU 刚开启、内核仍在 idmap (VA==PA 物理地址) 上跑, 此时还没有高 VA 映射, 所以它必须是一段 位置无关代码 (PI, position-independent)。

4.1 为什么叫 __pi_ 前缀？

// ============ __pi_ 前缀的来源 ============ arch/arm64/kernel/pi/Makefile: KBUILD_CFLAGS += -fpie // 编译成位置无关代码 (PIE) $(obj)/%.pi.o: OBJCOPYFLAGS := --prefix-symbols=__pi_ $(obj)/%.pi.o: $(obj)/%.o objcopy --prefix-symbols=__pi_ ... // 所有符号加 __pi_ 前缀 early_map_kernel --编译--> map_kernel.pi.o --objcopy--> __pi_early_map_kernel (PIE 版本) head.S 里 bl __pi_early_map_kernel 就是跳转到这个 PIE 版本。 因为此刻还没有绝对地址可用, 只能用 PC-relative 寻址。
💡 注意: 之前关于 "重新创建 idmap" 的说法不准确。 __pi_early_map_kernel 的核心任务是构建内核的高 VA 页表 (TTBR1)并完成重定位, 而不是重建 idmap。重建 idmap 只在 CONFIG_ARM64_LPA2 且 VA 位宽足够时才发生 (remap_idmap_for_lpa2), 属于特例。旧的 idmap 仍在 TTBR0 里继续服务, 直到 br x8 跳走后。

4.2 early_map_kernel 主流程 (map_kernel.c:241-289)

asmlinkage void __init early_map_kernel(u64 boot_status, phys_addr_t fdt)
{
    static char const chosen_str[] __initconst = "/chosen";
    u64 va_base, pa_base = (u64)&_text;            // pa_base = 内核当前物理基址
    u64 kaslr_offset = pa_base % MIN_KIMG_ALIGN;    // 低 21bit 来自物理摆放 (2MB 对齐)
    int root_level = 4 - CONFIG_PGTABLE_LEVELS;
    int va_bits = VA_BITS;
    int chosen;
    void *fdt_mapped = map_fdt(fdt);                // ① 先把 FDT 映射进 idmap

    memset(__bss_start, 0,                           // ② 清 BSS + 清 init 页表区域
           (char *)init_pg_end - (char *)__bss_start);

    chosen = fdt_path_offset(fdt_mapped, chosen_str); // ③ 找 /chosen 节点
    init_feature_override(boot_status, fdt_mapped, chosen); // 解析命令行 feature override

    if (IS_ENABLED(CONFIG_ARM64_64K_PAGES) && !cpu_has_lva())
        va_bits = VA_BITS_MIN;                       // ④ 降级 VA 位宽
    else if (IS_ENABLED(CONFIG_ARM64_LPA2) && !cpu_has_lpa2()) {
        va_bits = VA_BITS_MIN;
        root_level++;
    }

    if (va_bits > VA_BITS_MIN)
        sysreg_clear_set(tcr_el1, TCR_T1SZ_MASK, TCR_T1SZ(va_bits)); // ⑤ 设定 TCR_EL1.T1SZ

    if (IS_ENABLED(CONFIG_RANDOMIZE_BASE)) {         // ⑥ KASLR 偏移
        u64 kaslr_seed = kaslr_early_init(fdt_mapped, chosen);
        if (kaslr_seed && kaslr_requires_kpti())
            arm64_use_ng_mappings = ng_mappings_allowed();
        kaslr_offset |= kaslr_seed & ~(MIN_KIMG_ALIGN - 1);
    }

    if (IS_ENABLED(CONFIG_ARM64_LPA2) && va_bits > VA_BITS_MIN)
        remap_idmap_for_lpa2();                      // ⑦ LPA2 专用: 重建 idmap

    va_base = KIMAGE_VADDR + kaslr_offset;           // ⑧ 内核最终 VA 基址
    map_kernel(kaslr_offset, va_base - pa_base, root_level); // ⑨ ★ 核心映射 + 重定位
}
步动作为什么此时做
①map_fdt(fdt)FDT 此刻还在物理地址, 必须先临时映射进 idmap, 后面的 fdt_path_offset/fdt_getprop 才能访问它
②memset(__bss_start, 0, ...)清 BSS 段 + 清 init_pg_dir .. init_pg_end 这段临时页表区, 保证页表从零开始干净
③init_feature_override解析 /chosen/bootargs 里的 CPU 特性覆盖 (类似 id_aa64* 模拟), 影响后续页表属性
④⑤调整 va_bits / T1SZ64K 页/LPA2 不支持时降级; 据此设 TCR_EL1.T1SZ, 决定高 VA 翻译的地址宽度
⑥kaslr_early_init从 /chosen/kaslr-seed (或 RNDR 指令) 拿随机种子, 算 KASLR 偏移高位。
kaslr_offset = 物理摆放低位 | 种子高位
⑦remap_idmap_for_lpa2仅 LPA2: LPA2 打开后 PTE 的 [9:8] 位含义改变, 需重建一个 LPA2 兼容的临时 idmap
⑧⑨map_kernel(...)★ 核心: 六段映射到高 VA + relocate_kernel 重定位 + 安装 TTBR1

4.3 map_kernel：六段映射 + 安装 TTBR1 (map_kernel.c:39-139)

static void __init map_kernel(u64 kaslr_offset, u64 va_offset, int root_level)
{
    bool twopass = IS_ENABLED(CONFIG_RELOCATABLE);   // 需要重定位 → 两遍映射
    phys_addr_t pgdp = (phys_addr_t)init_pg_dir + PAGE_SIZE; // 次级页表分配指针
    pgprot_t text_prot = PAGE_KERNEL_ROX;            // 代码: 只读可执行
    pgprot_t data_prot = PAGE_KERNEL;                // 数据: 可读可写
    pgprot_t prot = twopass ? data_prot : text_prot; // 第一遍先 RW, 重定位后再 ROX

    // ---- 六个内核段逐个映射 (VA = PA + va_offset) ----
    map_segment(init_pg_dir, &pgdp, va_offset, _text, _stext, data_prot, false, root_level);
    map_segment(init_pg_dir, &pgdp, va_offset, _stext, _etext, prot, !twopass, root_level);
    map_segment(init_pg_dir, &pgdp, va_offset, __start_rodata, __inittext_begin, data_prot, false, root_level);
    map_segment(init_pg_dir, &pgdp, va_offset, __inittext_begin, __inittext_end, prot, false, root_level);
    map_segment(init_pg_dir, &pgdp, va_offset, __initdata_begin, __initdata_end, data_prot, false, root_level);
    map_segment(init_pg_dir, &pgdp, va_offset, _data, _end, data_prot, true, root_level);
    dsb(ishst);

    idmap_cpu_replace_ttbr1((phys_addr_t)init_pg_dir); // ★ 临时 TTBR1 = init_pg_dir

    if (twopass) {
        if (IS_ENABLED(CONFIG_RELOCATABLE))
            relocate_kernel(kaslr_offset);            // ★ 重定位: 修正所有绝对指针
        if (enable_scs)
            scs_patch(...);                           // SCS 影子调用栈补丁
        unmap_segment(init_pg_dir, va_offset, _stext, _etext, root_level);
        __tlbi(vmalle1);                              // 先 unmap 代码段
        map_segment(init_pg_dir, NULL, va_offset, _stext, _etext, text_prot, true, root_level);
        map_segment(init_pg_dir, NULL, va_offset, __inittext_begin, __inittext_end, text_prot, false, root_level);
        // 第二遍: 把代码段重新映射为 ROX
    }

    memcpy((void *)swapper_pg_dir + va_offset, init_pg_dir, PAGE_SIZE); // 拷贝根页表到最终位置
    dsb(ishst);
    idmap_cpu_replace_ttbr1((phys_addr_t)swapper_pg_dir); // ★ 最终 TTBR1 = swapper_pg_dir
}
内核段起始 → 结束保护说明
镜像头_text → _stextRW (NX)PE/COFF 头等不可执行数据, 启动后不再执行
代码段_stext → _etextROX内核主体代码 (两遍时先 RW 重定位, 再 ROX)
只读数据__start_rodata → __inittext_beginRWrodata, 此时仍 RW, 之后 mark_rodata_ro 才改 RO
init 代码__inittext_begin → __inittext_endROX.init.text, 启动后释放
init 数据__initdata_begin → __initdata_endRW.init.data
主数据_data → _endRW主数据段 + BSS
💡 为什么重定位要两遍 (twopass)？ CONFIG_RELOCATABLE 下, 内核要执行 relocate_kernel 修正绝对指针, 而这会改写代码段里的数据 (relocation 位置)。所以第一遍把代码段临时映射成 RW, 重定位完成后再 unmap + 重映成 ROX。非 RELOCATABLE 内核没有这一步, 一遍 ROX 直接到位。

4.4 relocate_kernel：重定位 (relocate.c:15-64)

void __init relocate_kernel(u64 offset)
{
    // ① RELA 表: R_AARCH64_RELATIVE 类型
    for (rela = rela_start; rela < rela_end; rela++)
        *(u64 *)(rela->r_offset + offset) = rela->r_addend + offset;

    // ② RELR 表: 压缩格式, 每 64bit 位图可表达 63 个相对重定位
    for (relr = relr_start; relr < relr_end; relr++) {
        if ((*relr & 1) == 0) {          // 地址项: 一个重定位
            place = (u64 *)(*relr + offset);
            *place++ += offset;
        } else {                          // 位图项: 后续连续地址逐个 +offset
            for (p = place, r = *relr >> 1; r; p++, r >>= 1)
                if (r & 1) *p += offset;
            place += 63;
        }
    }
}

内核是 -fpie 编译的 PIE, 内部所有取绝对地址的地方都记录在 RELA/RELR 重定位表里。 把它们统一 + kaslr_offset, 内核就能在新 VA 地址 (KIMAGE_VADDR + kaslr_offset) 正确运行。

4.5 KASLR offset 的计算

// ============ kaslr_offset 的两个来源 ============ kaslr_offset = pa_base % MIN_KIMG_ALIGN // 低位: 物理摆放 (2MB 对齐的余量) | kaslr_seed & ~(MIN_KIMG_ALIGN-1) // 高位: /chosen/kaslr-seed 种子 va_base = KIMAGE_VADDR + kaslr_offset // 内核运行时 VA 基址 va_offset = va_base - pa_base // 映射偏移量 map_segment: VA = PA + va_offset // 每段都套这个偏移 ★ 关键: 有 KASLR 时, 内核符号运行时地址 = 链接地址 + kaslr_offset
⚠️ 对 GDB 断点的直接影响: 如果启用了 KASLR (QEMU 提供 kaslr-seed 且未加 nokaslr), 那么 start_kernel 等的运行时虚拟地址会偏移一个随机的 kaslr_offset。 这时按 vmlinux 链接地址设的断点会打不中。调试时建议在 cmdline 加 nokaslr, 或读取运行时的 kimage_voffset 后手动加偏移。

4.6 idmap_cpu_replace_ttbr1：为什么必须在 idmap.text

// arch/arm64/mm/proc.S — 位于 .idmap.text 段
SYM_TYPED_FUNC_START(idmap_cpu_replace_ttbr1)
    __idmap_cpu_set_reserved_ttbr1 x1, x3   // 先切到 reserved 空页表 (防空窗)
    offset_ttbr1 x0, x3                      // x0 (物理) → TTBR1 格式
    msr     ttbr1_el1, x0                    // ★ 写 TTBR1_EL1
    isb
    ret
SYM_FUNC_END(idmap_cpu_replace_ttbr1)

这个函数在 map_kernel 里被调用两次 (先 init_pg_dir 后 swapper_pg_dir)。 它必须放在 .idmap.text: 因为写 TTBR1_EL1 的瞬间, 若当前指令取自 TTBR1 翻译的地址, 就会因页表切换出现"正在执行的代码消失"的取指异常。放在 idmap (由 TTBR0 翻译, VA==PA) 里执行, 切换 TTBR1 不影响自身取指, 安全。

4.7 收尾：memcpy 拷贝 PGD + 切换正式 TTBR1

// map_kernel.c:136-138 — map_kernel 的最后三行
memcpy((void *)swapper_pg_dir + va_offset, init_pg_dir, PAGE_SIZE);
dsb(ishst);
idmap_cpu_replace_ttbr1((phys_addr_t)swapper_pg_dir);

这是 map_kernel 的收尾：把建好的根页表从临时位置搬到正式位置，再切到正式页表。

swapper_pg_dirRO 区(vmlinux.lds.S:234)一页 4KB正式 PGD，后续 map_mem() 等全引用它

为什么只拷 PAGE_SIZE 一页就够了

// ============ PGD 拷贝的语义 ============ init_pg_dir (BSS, 永久): [页0] PGD ← memcpy 只拷这一页 [页1] PUD ← 留在原地 [页2] PMD ← 留在原地 [页3] PTE ← 留在原地 ... swapper_pg_dir (RO 区, 4KB): [页0] PGD (拷贝来的) → 指向 init_pg_dir 里的下级页表 PGD 条目里存的是"下级页表的物理地址"， 拷了 PGD，下级页表就自动被新 PGD 指到了。

关键：下级页表留在 init_pg_dir 的 BSS 区域里，BSS 是永久内存， 不会像 .init.text/.init.data 那样被 free_initmem 释放。

+ va_offset 是什么意思

此刻代码仍跑在 idmap(物理地址)上，swapper_pg_dir 符号被引用得到的是物理地址。 + va_offset 把它换算成最终内核映射里的虚拟地址（link 地址 + kaslr_offset）， 所以 memcpy 的目标走 TTBR1（已装好 init_pg_dir 的高 VA 映射），落到正确的物理页。

💡 同一个符号两种用法：
(void *)swapper_pg_dir + va_offset 是访问内存 → 用虚拟地址（走 MMU 翻译）；
(phys_addr_t)swapper_pg_dir 是装进 TTBR1_EL1 → 用物理地址 （该函数内部 offset_ttbr1 会把物理地址转成 TTBR 格式）。

dsb(ishst) 为什么必须加

MMU 的页表走读不经过 CPU 的 store buffer。如果 PGD 的拷贝还躺在 store buffer 里没落内存， 下一句切 TTBR1 后硬件就会读到旧页表。所以先 dsb(ishst) 强制写穿。

第二次(最终)安装 TTBR1

这是第二次切 TTBR1（第一次在 map_kernel.c:99 装 init_pg_dir）。这次换成正式的 swapper_pg_dir。 两次都是调 proc.S 的 idmap_cpu_replace_ttbr1，它必须放 .idmap.text：切 TTBR1 瞬间自身取指不能依赖 TTBR1。

4.8 完成后的页表布局

// ============ __pi_early_map_kernel 返回后的页表 ============ TTBR0_EL1 → __pi_init_idmap_pg_dir (旧 idmap, VA=PA, 仍在服务) TTBR1_EL1 → swapper_pg_dir (正式内核页表) swapper_pg_dir 内容 (va_offset = KIMAGE_VADDR + kaslr_offset - pa_base): KIMAGE_VADDR + kaslr_offset + 0x00000000 → PA 0x40200000 (代码 _stext) KIMAGE_VADDR + kaslr_offset + 0x01000000 → PA 0x41000000 (rodata) ... (六段映射全部就位) ... 返回后 __primary_switch 执行: ldr x8, =__primary_switched // 高 VA (已按 kaslr_offset 修正) br x8 // ★ 跳到虚拟地址世界!

5. __primary_switched → start_kernel (head.S:220-248)

SYM_FUNC_START_LOCAL(__primary_switched)
    adr_l   x4, init_task                  // ① 获取 init_task 地址
    init_cpu_task x4, x5, x6               // ② 初始化 CPU 任务上下文

    adr_l   x8, vectors                    // ③ 安装异常向量表
    msr     vbar_el1, x8                   //    (现在 VA 可用了)
    isb

    stp     x29, x30, [sp, #-16]!          // ④ 标准 C 函数序言
    mov     x29, sp

    str_l   x21, __fdt_pointer, x5         // ⑤ 全局保存 FDT 指针

    adrp    x4, _text                      // ⑥ ★ 计算 kimage_voffset
    sub     x4, x4, x0                     //     = VA(_text) - PA(_text)
    str_l   x4, kimage_voffset, x5         //     保存到全局变量

    mov     x0, x20
    bl      set_cpu_boot_mode_flag         // ⑦ 记录 boot CPU mode

#if defined(CONFIG_KASAN_GENERIC) || defined(CONFIG_KASAN_SW_TAGS)
    bl      kasan_early_init               // ⑧ KASAN 早期初始化 (可选)
#endif
    mov     x0, x20
    bl      finalise_el2                   // ⑨ 尝试启用 VHE (EL2 模式)

    ldp     x29, x30, [sp], #16            // ⑩ 恢复栈帧
    bl      start_kernel                   // ⑪ ★ 进入 C 代码主函数!
    ASM_BUG()                              // ⑫ 永远不会执行 (start_kernel 不返回)
SYM_FUNC_END(__primary_switched)

逐行关键分析

行做什么为什么此时做
①②设置 init_task → sp_el0, 初始化任务栈C 代码需要 current 宏工作, current 依赖 sp_el0
③msr vbar_el1, vectors★ 安装异常向量表。此前没有向量表!
如果在 MMU on 后、VBAR 设置前发生异常 → 不可预知的行为
⑤str_l x21, __fdt_pointerFDT 地址写入全局变量, setup_arch() 将通过它解析设备树
⑥sub x4, x4, x0★ 计算 kimage_voffset = VA - PA
这是内核最重要的运行时常量之一:
__kimg_to_phys(addr) = addr - kimage_voffset
⑦set_cpu_boot_mode_flag把 x20 (cpu_boot_mode) 写入 __boot_cpu_mode[] 数组
⑨finalise_el2如果硬件支持 VHE, 启用之, 让内核跑在 EL2
⑪bl start_kernel★ C 代码的入口。参数: x0=0 (传给 init/main.c:start_kernel)

⑥ kimage_voffset 的计算

// ============ 关键公式 ============ adrp x4, _text // x4 = VA of _text = 0xffff800080000000 // (adrp 在 VA 空间下解析, 得到链接器分配的 VA) sub x4, x4, x0 // x0 = PA of _text (从 __primary_switch 传入) // x4 = VA - PA = kimage_voffset // = 0xffff800080000000 - 0x40200000 // = 0xffff80003FE00000 后续: 任意内核符号 S 的物理地址 = S - kimage_voffset
💡 为什么此时 VA 可用？ 因为 __pi_early_map_kernel 已经在 TTBR1 中建立了内核高 VA 的映射。CPU 发出的高地址能正常翻译为正确的 PA。计算 kimage_voffset 时, adrp x4, _text 通过 TTBR1 翻译拿到了正确的链接时高 VA 值 — 因为我们现在在 VA 空间里运行。
