1. 概述：primary_entry 的最后一条指令

// ============ primary_entry 完整调用链 ============ primary_entry: bl record_mmu_state // x19 = boot MMU flag bl preserve_boot_args // 保存 FDT + boot_args ... 创建 idmap ... bl __pi_create_init_idmap // 构建 identity mapping 页表 ... cache 维护 ... mov x0, x19 bl init_kernel_el // EL 切换, 关 MMU → w0=cpu_boot_mode mov x20, x0 // x20 = cpu_boot_mode bl __cpu_setup // 配置 TCR/MAIR → x0=SCTLR_MMU_ON b __primary_switch // ★ 最后一条指令: 无条件跳转 // 此后 primary_entry 不再返回 __primary_switch: bl __enable_mmu // ★① 开启 MMU (idmap 生效) mov sp, early_init_stack // ② 切到新栈 (现在是 VA 了) bl __pi_early_map_kernel // ③ 重定位内核到高 VA ldr x8, =__primary_switched // ④ 加载绝对 VA (高地址) br x8 // ⑤ 跳转到虚拟地址世界!

从 b __primary_switch 到 bl start_kernel，总共约 20 条指令，完成了 ARM64 内核启动中最重要的三件事：开启 MMU、重定位内核、进入 C 代码。

2. __primary_switch 逐行分析 (head.S:508-523)

SYM_FUNC_START_LOCAL(__primary_switch)
    adrp    x1, reserved_pg_dir            // ① x1 = 空页表 (全零 PG)
    adrp    x2, __pi_init_idmap_pg_dir     // ② x2 = idmap PGD
    bl      __enable_mmu                   // ③ ★ 开启 MMU!

    // ===== 以下是 MMU ON 之后 =====
    adrp    x1, early_init_stack           // ④ 获取新栈地址 (现在是 VA!)
    mov     sp, x1                         // ⑤ 切到新栈
    mov     x29, xzr                       // ⑥ FP=0 (新栈帧起点)
    mov     x0, x20                        // ⑦ x0 = cpu_boot_mode
    mov     x1, x21                        // ⑧ x1 = FDT 物理地址
    bl      __pi_early_map_kernel          // ⑨ 重定位 + 映射内核

    ldr     x8, =__primary_switched        // ⑩ x8 = __primary_switched 的绝对 VA
    adrp    x0, KERNEL_START               // ⑪ x0 = _text 的物理地址
    br      x8                             // ⑫ 跳转到虚拟地址!
行指令含义
①②adrp x1, reserved_pg_dir
adrp x2, __pi_init_idmap_pg_dir		x1 = reserved 空页表 (全 0, 用于 TTBR1 切换)
x2 = 我们刚刚构建的 idmap PGD
③bl __enable_mmu★ 开启 MMU! 见下一章详解
④⑤adrp x1, early_init_stack; mov sp, x1重新设置栈指针。MMU ON 之前这行用的是物理地址，MMU ON 之后同一地址经过 idmap 翻译 → 结果相同
⑥mov x29, xzrFP=0, 新栈帧起点。GDB backtrace 的终止条件
⑦⑧mov x0, x20; mov x1, x21传递 cpu_boot_mode 和 FDT 给 __pi_early_map_kernel
⑨bl __pi_early_map_kernel映射内核到高 VA, 重定位页表, 设置 TTBR1
此后内核主体可以跑在高 VA 上了
⑩⑪⑫ldr x8, =__primary_switched
br x8		★ 这是从 idmap (物理地址) 到高 VA (虚拟地址) 的跳跃!
ldr x8, = 加载的是链接器分配的绝对 VA

关键：br x8 为什么能成功？

// ============ br x8 的时刻 ============ 执行 br x8 之前: PC = 某个物理地址 (通过 idmap VA=PA 翻译) 页表: TTBR0 = idmap PGD, TTBR1 = 内核正式 PGD 执行 br x8: x8 = 0xffff8000_xxxx_xxxx (__primary_switched 的链接 VA) CPU 发出地址 0xffff8000_xxxx_xxxx: → 这是高 VA 地址 → TCR_EL1.T1SZ 判定 → 走 TTBR1 翻译 → TTBR1 指向 __pi_early_map_kernel 刚建好的内核页表 → 翻译成功! CPU 跳到 __primary_switched 这一瞬间, CPU 从 "idmap 物理地址世界" 跳到了 "内核虚拟地址世界" 这是 ARM64 启动中最精彩的单条指令

3. __enable_mmu: MMU 开启的瞬间 (head.S:459-473)

SYM_FUNC_START(__enable_mmu)
    mrs     x3, ID_AA64MMFR0_EL1           // ① 读硬件特性寄存器
    ubfx    x3, x3, #ID_AA64MMFR0_EL1_TGRAN_SHIFT, 4
    cmp     x3, #ID_AA64MMFR0_EL1_TGRAN_SUPPORTED_MIN
    b.lt    __no_granule_support            // ② 4KB 粒度不支持? → 死循环
    cmp     x3, #ID_AA64MMFR0_EL1_TGRAN_SUPPORTED_MAX
    b.gt    __no_granule_support            // ③ 粒度编号超出范围? → 死循环
    phys_to_ttbr x2, x2                     // ④ 物理地址 → TTBR 格式
    msr     ttbr0_el1, x2                   // ⑤ 安装 idmap PGD 到 TTBR0
    load_ttbr1 x1, x1, x3                   // ⑥ 安装 reserved 空页表到 TTBR1

    set_sctlr_el1   x0                      // ⑦ ★ 写入 SCTLR_EL1 = MMU_ON!
                                            //    msr sctlr_el1, x0
                                            //    isb
                                            //    ic iallu     ← 刷 I-cache
                                            //    dsb nsh
                                            //    isb
    ret                                     // ⑧ ★ 返回, 但此时 MMU 已在翻译!
SYM_FUNC_END(__enable_mmu)

逐行解释

行含义
①②③安全检查：读 ID_AA64MMFR0_EL1.TGran4 确认硬件支持 4KB 页表粒度。不支持则跳入 __no_granule_support — 无限循环 (wfi/ b)
④phys_to_ttbr x2, x2：把 idmap PGD 的物理地址转换为 TTBR 格式。
对于 48-bit PA: 直接返回 PA (bits[47:1] ← PA[47:1], bit[0] 可能为 CnP)
对于 52-bit PA: 做位折叠 (类似 __phys_to_pte_val)
⑤msr ttbr0_el1, x2：安装 idmap PGD 到 TTBR0。
★ 此后硬件遍历页表时就会读我们的 idmap 页表了
⑥load_ttbr1 x1, x1, x3：安装 reserved 空页表到 TTBR1。
reserved_pg_dir 是一页全 0 的物理页 — 任何 TTBR1 翻译都产生 translation fault
⑦set_sctlr_el1 x0：这条宏做了三件事：
1. msr sctlr_el1, x0 — 写入 SCTLR (M=1!) ★ MMU 在此刻生效
2. isb — 同步
3. ic iallu + dsb nsh + isb — 全刷 I-cache (清除 speculative fetch 的旧指令)
⑧ret：返回 __primary_switch。
★ 这条 ret 本身也是经 MMU 翻译的!
取指: PC → idmap 查表 → PA=PC → 正确!
这就是 idmap 存在的全部意义

set_sctlr_el1 的展开

// arch/arm64/include/asm/assembler.h:720-731
.macro set_sctlr, sreg, reg
    msr     \sreg, \reg          // 写 SCTLR_EL1 = INIT_SCTLR_EL1_MMU_ON
    isb                          // ★ 指令同步屏障
    ic      iallu                // 全刷 I-cache 到 PoU
    dsb     nsh
    isb
.endm

ic iallu 刷新整个 I-cache 到 PoU (Point of Unification)。因为 MMU 开启改变了翻译环境，之前 speculative fetch 进来的旧指令可能基于错误的翻译。不刷 I-cache 可能导致 CPU 执行到旧翻译下的错误指令。

MMU 开启的精确时刻

// ============ msr sctlr_el1, x0 前后的变化 ============ msr sctlr_el1, x0 之前: CPU 取指: PA = PC (MMU off, 无翻译) 地址空间: 物理地址 msr sctlr_el1, x0 之后 (M 位从 0 → 1): CPU 取指: PA = idmap_walk(PC) = PC (VA==PA, identity 翻译) 地址空间: 仍然是物理地址 (因为 idmap 保证 VA==PA) TTBR0_EL1 指向的 idmap PGD 开始被硬件使用 TCR_EL1 的 T0SZ/TG0/SH0 等配置开始生效

4. MMU ON 之后：__pi_early_map_kernel (head.S:516-518)

    mov     x0, x20                        // cpu_boot_mode
    mov     x1, x21                        // FDT 物理地址
    bl      __pi_early_map_kernel          // 映射 + 重定位内核!

这个 PI 函数 (在 map_kernel.c 中) 做三件事：

  1. 构建正式页表 — 在 init_pg_dir 中创建内核的高 VA (0xffff...) 映射
  2. 重新创建 idmap — 用正确的 PA 重建 identity mapping (此时已知 kimage_voffset)
  3. 安装 TTBR1 — 把 init_pg_dir 设为 TTBR1_EL1
💡 为什么需要重新创建 idmap？ 第一次 create_init_idmap 用的 VA==PA 是通过 -fpie PC-relative 拿到的物理地址。现在 __pi_early_map_kernel 里可以精确计算 kimage_voffset, 创建正确的 "VA(高) → PA(实际)" 映射。旧的 idmap PGD 仍然在 TTBR0 中, 但在 __primary_switched 之后会被替换。

完成后的页表布局

// ============ __pi_early_map_kernel 后的页表 ============ TTBR0_EL1 → __pi_init_idmap_pg_dir (旧的 idmap) VA=PA, 仍然有效 TTBR1_EL1 → init_pg_dir (新的正式页表) 0xffff8000_80000000 → PA 0x40200000 (内核代码) 0xffff8000_81000000 → PA 0x41000000 (内核数据) ... 现在内核主体可以在高 VA 下运行了!

5. __primary_switched → start_kernel (head.S:220-248)

SYM_FUNC_START_LOCAL(__primary_switched)
    adr_l   x4, init_task                  // ① 获取 init_task 地址
    init_cpu_task x4, x5, x6               // ② 初始化 CPU 任务上下文

    adr_l   x8, vectors                    // ③ 安装异常向量表
    msr     vbar_el1, x8                   //    (现在 VA 可用了)
    isb

    stp     x29, x30, [sp, #-16]!          // ④ 标准 C 函数序言
    mov     x29, sp

    str_l   x21, __fdt_pointer, x5         // ⑤ 全局保存 FDT 指针

    adrp    x4, _text                      // ⑥ ★ 计算 kimage_voffset
    sub     x4, x4, x0                     //     = VA(_text) - PA(_text)
    str_l   x4, kimage_voffset, x5         //     保存到全局变量

    mov     x0, x20
    bl      set_cpu_boot_mode_flag         // ⑦ 记录 boot CPU mode

#if defined(CONFIG_KASAN_GENERIC) || defined(CONFIG_KASAN_SW_TAGS)
    bl      kasan_early_init               // ⑧ KASAN 早期初始化 (可选)
#endif
    mov     x0, x20
    bl      finalise_el2                   // ⑨ 尝试启用 VHE (EL2 模式)

    ldp     x29, x30, [sp], #16            // ⑩ 恢复栈帧
    bl      start_kernel                   // ⑪ ★ 进入 C 代码主函数!
    ASM_BUG()                              // ⑫ 永远不会执行 (start_kernel 不返回)
SYM_FUNC_END(__primary_switched)

逐行关键分析

行做什么为什么此时做
①②设置 init_task → sp_el0, 初始化任务栈C 代码需要 current 宏工作, current 依赖 sp_el0
③msr vbar_el1, vectors★ 安装异常向量表。此前没有向量表!
如果在 MMU on 后、VBAR 设置前发生异常 → 不可预知的行为
⑤str_l x21, __fdt_pointerFDT 地址写入全局变量, setup_arch() 将通过它解析设备树
⑥sub x4, x4, x0★ 计算 kimage_voffset = VA - PA
这是内核最重要的运行时常量之一:
__kimg_to_phys(addr) = addr - kimage_voffset
⑦set_cpu_boot_mode_flag把 x20 (cpu_boot_mode) 写入 __boot_cpu_mode[] 数组
⑨finalise_el2如果硬件支持 VHE, 启用之, 让内核跑在 EL2
⑪bl start_kernel★ C 代码的入口。参数: x0=0 (传给 init/main.c:start_kernel)

⑥ kimage_voffset 的计算

// ============ 关键公式 ============ adrp x4, _text // x4 = VA of _text = 0xffff800080000000 // (adrp 在 VA 空间下解析, 得到链接器分配的 VA) sub x4, x4, x0 // x0 = PA of _text (从 __primary_switch 传入) // x4 = VA - PA = kimage_voffset // = 0xffff800080000000 - 0x40200000 // = 0xffff80003FE00000 后续: 任意内核符号 S 的物理地址 = S - kimage_voffset
💡 为什么此时 VA 可用？ 因为 __pi_early_map_kernel 已经在 TTBR1 中建立了内核高 VA 的映射。CPU 发出的高地址能正常翻译为正确的 PA。计算 kimage_voffset 时, adrp x4, _text 通过 TTBR1 翻译拿到了正确的链接时高 VA 值 — 因为我们现在在 VA 空间里运行。

6.4 核心验证：idmap 让 MMU 开启瞬间不崩溃

// ============ ret 指令的翻译过程 ============ __enable_mmu 中的 ret: PC = 0x4163c834 (物理地址, 在 idmap 段) set_sctlr_el1 刚执行完 → SCTLR_EL1.M = 1 TTBR0_EL1 = idmap PGD (VA=PA 映射) TCR_EL1.T0SZ = 48 CPU 取指: 发出 VA = 0x4163c834 → 查 TTBR0 (VA < 2^48, 走 TTBR0) → PGD[0] → PUD[..] → PMD[..] → Block entry: OA = 0x4163c834 >> 21 = ... → PA = 0x4163c834 → VA == PA! idmap 生效! → 取指成功! ret 的下一条指令被正确执行 如果没有 idmap (或 idmap 覆盖范围不够): → 翻译失败 → Prefetch Abort → 内核崩溃
